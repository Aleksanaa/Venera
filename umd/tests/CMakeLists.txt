cmake_minimum_required(VERSION 3.24)
project(venera-umd-log-tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
add_compile_options(-Wall -Wextra)
enable_testing()

if(WIN32)
  foreach(api IN ITEMS 11 12)
    add_library(log-d3d${api} SHARED log-module.cpp ../common/log.cpp)
    target_compile_definitions(log-d3d${api} PRIVATE LOG_COMPONENT="venera_d3d${api}")
    target_link_libraries(log-d3d${api} PRIVATE advapi32)
    target_link_options(log-d3d${api} PRIVATE -static -static-libgcc -static-libstdc++)
    set_target_properties(log-d3d${api} PROPERTIES PREFIX "")
  endforeach()
  add_executable(log-capture log-capture.cpp)
  target_link_libraries(log-capture PRIVATE advapi32)
  target_link_options(log-capture PRIVATE -static -static-libgcc -static-libstdc++)
  add_test(NAME log-capture COMMAND log-capture)
else()
  find_package(Threads REQUIRED)
  add_executable(log-behavior log-behavior.cpp ../common/log.cpp)
  target_include_directories(log-behavior PRIVATE winapi)
  target_link_libraries(log-behavior PRIVATE Threads::Threads)
  add_test(NAME log-behavior-d3d11 COMMAND log-behavior venera_d3d11 5)
  add_test(NAME log-behavior-d3d12 COMMAND log-behavior venera_d3d12 6)
  add_test(NAME log-registration-fallback COMMAND log-behavior registration)
  add_test(NAME log-process-time-fallback COMMAND log-behavior process-time)
  add_test(NAME log-verbose-setting COMMAND log-behavior verbose-off)
  add_test(NAME log-file-retry COMMAND log-behavior file-retry)
endif()
