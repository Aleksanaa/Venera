cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(venera-setup-tests LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
endif()

enable_testing()
add_test(NAME setup-viosock-payload COMMAND "${CMAKE_COMMAND}" -P "${CMAKE_CURRENT_SOURCE_DIR}/viosock-payload.cmake")
add_executable(setup-components-test components.cpp)
add_test(NAME setup-components COMMAND setup-components-test)
add_executable(setup-viosock-state-test viosock-state.cpp)
add_test(NAME setup-viosock-state COMMAND setup-viosock-state-test)

if(WIN32)
  target_link_options(setup-components-test PRIVATE -static -static-libgcc -static-libstdc++)
  add_executable(setup-uninstaller-copy-test uninstaller-copy.cpp ../uninstaller-copy.cpp)
  target_compile_definitions(setup-uninstaller-copy-test PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN)
  target_link_options(setup-uninstaller-copy-test PRIVATE -static -static-libgcc -static-libstdc++)
  target_link_options(setup-viosock-state-test PRIVATE -static -static-libgcc -static-libstdc++)
  add_test(NAME setup-uninstaller-copy COMMAND setup-uninstaller-copy-test)
endif()
